! Lab example: Cisco Catalyst access/distribution switch baseline (IOS).
! Replace addresses and secrets. Never publish real credentials.
hostname SW-CORE-01
service password-encryption
no ip http server
no ip http secure-server
enable secret <STRONG-SECRET>
username netadmin privilege 15 secret <STRONG-SECRET>
ip domain-name lab.example.rw
crypto key generate rsa modulus 2048
ip ssh version 2
!
banner motd ^Authorized access only. Activity is monitored.^
!
vlan 10
 name SERVERS
vlan 20
 name FINANCE
vlan 30
 name USERS
vlan 99
 name MANAGEMENT
!
interface vlan 99
 ip address 10.10.99.2 255.255.255.0
 no shutdown
ip default-gateway 10.10.99.1
!
! Access port with port-security (end-user)
interface range fastEthernet0/1 - 20
 description USER-PORTS
 switchport mode access
 switchport access vlan 30
 switchport port-security
 switchport port-security maximum 2
 switchport port-security violation restrict
 spanning-tree portfast
 spanning-tree bpduguard enable
!
! Server port
interface fastEthernet0/21
 description APP-SERVER-01
 switchport mode access
 switchport access vlan 10
!
! Trunk to router/firewall
interface gigabitEthernet0/1
 description TRUNK-TO-FIREWALL
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,20,30,99
 switchport nonegotiate
!
! Unused ports shut down in a black-hole VLAN
vlan 999
 name UNUSED
interface range fastEthernet0/22 - 24
 switchport access vlan 999
 shutdown
!
! Management hardening
ip access-list standard MGMT-ONLY
 permit 10.10.99.0 0.0.0.255
line vty 0 4
 access-class MGMT-ONLY in
 transport input ssh
 login local
 exec-timeout 10 0
!
! Time and logging
ntp server 10.10.99.10
logging host 10.10.99.20
logging trap informational
service timestamps log datetime msec
!
spanning-tree mode rapid-pvst
end
write memory
